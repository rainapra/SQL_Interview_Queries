MySQL
WITH RECURSIVE DateRange AS (
  -- Base case: start from each employee's start date
  SELECT employee_id, start_date AS work_date, end_date
  FROM projects
  
  UNION ALL
  
  -- Recursive case: keep adding one day until the end date
  SELECT employee_id, work_date + INTERVAL 1 DAY, end_date
  FROM DateRange
  WHERE work_date < end_date
)
-- Count distinct work dates for each employee
SELECT employee_id, 
       COUNT(DISTINCT work_date) AS unique_days_worked
FROM DateRange
GROUP BY employee_id
ORDER BY employee_id;

FROM LogInfo l1
JOIN LogInfo l2 ON l1.account_id = l2.account_id
WHERE l1.ip_address <> l2.ip_address
AND (l1.login BETWEEN l2.login AND l2.logout
     OR l1.logout BETWEEN l2.login AND l2.logout);

from LogInfo l1, LogInfo l2
where l1.account_id = l2.account_id
and l1.ip_address <> l2.ip_address
and( l1.login = l2.login or l1.logout = l2.logout or l1.login = l2.logout)
or (l1.login < l2.login and l1.logout > l2.login)

FROM LogInfo
GROUP BY account_id, ip_address
HAVING COUNT(DISTINCT ip_address) > 1 
AND EXISTS (
    SELECT 1
    FROM LogInfo AS l2

    AND LogInfo.ip_address <> l2.ip_address

       E1.name,
       COUNT(E2.employee_id) AS reports_count,
